# System Metrics Extension Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A VS Code extension with its own activity-bar icon that shows Task Manager-style live graphs and values for CPU, memory, GPU (per adapter, with dedicated/shared memory), disk and network.

**Architecture:** Plain CommonJS, no build step. `src/metrics.js` collects cross-platform data through `systeminformation` and schedules two polling tiers (1s/10s). On Windows, `src/winCounters.js` runs one long-lived PowerShell (`src/counters.ps1`) that streams performance counters plus a DXGI adapter list as JSON lines. `src/extension.js` hosts a webview view; `media/view.js` renders snapshots with hand-drawn canvas graphs.

**Tech Stack:** Node (VS Code extension host), `systeminformation` ^5.33.15, Windows PowerShell 5.1 + `Get-Counter` + DXGI via `Add-Type`, `node:test`, `@vscode/vsce` ^4.0.0.

**Spec:** `docs/superpowers/specs/2026-10-06-system-metrics-extension-design.md`

## Global Constraints

- Plain JavaScript (CommonJS), no TypeScript, no bundler, no build step.
- Only runtime dependency: `systeminformation`. No chart library.
- Fast tier every 1000 ms; slow tier every 10000 ms; graph history 60 points.
- `null` means N/A everywhere; the webview renders `N/A` and never throws on a missing field.
- PowerShell child: restart 5000 ms after an unexpected exit; give up after 3 consecutive failures and post `{ gpu: null, disk: { active: null }, net: { rxSec: null, txSec: null } }`.
- CPU temperature is always `null` on Windows.
- Webview CSP: `default-src 'none'`, styles from `webview.cspSource`, scripts only by nonce.
- Everything in English. **Subagents never commit, never create or switch branches.** The single commit for this plan happens in Task 6, after the user approves it.
- Windows shell: run commands from `D:\GIT\SystemMetricsExtension`; quote paths.

## Deviations from the spec (decided while planning)

- The PowerShell script lives in its own file `src/counters.ps1` (spawned with `-File`) instead of a string inside `winCounters.js` — avoids escaping `$` and here-strings inside JS.
- `poll`/`startSources` (the timer scheduling) live in `src/metrics.js` so they are testable without VS Code.
- Disk activity % on macOS/Linux is `null`: `systeminformation` has no busy-% there (`disksIO()` gives ops/sec). Marked with a `ponytail:` comment.
- Known limitation: `Get-Counter` takes localized counter names, so on a non-English Windows display language the counter loop yields `null` GPU/disk/network. The dev machine reports English names.

```js
// Shared by the webview (as window.fmt) and by node tests (as a CommonJS module).
(function (root) {
  const UNITS = ['B', 'KB', 'MB', 'GB', 'TB'];

  function bytes(n) {
    if (n == null) return 'N/A';
    let i = 0;
    while (n >= 1024 && i < UNITS.length - 1) {
      n /= 1024;
      i++;
    }
    return `${n.toFixed(i === 0 ? 0 : 1)} ${UNITS[i]}`;
  }

  function rate(n) {
    return n == null ? 'N/A' : `${bytes(n)}/s`;
  }

  function pct(n) {
    return n == null ? 'N/A' : `${n.toFixed(1)}%`;
  }

  function uptime(sec) {
    if (sec == null) return 'N/A';
    sec = Math.floor(sec);
    const pad = (x) => String(x).padStart(2, '0');
    const d = Math.floor(sec / 86400);
    const h = Math.floor((sec % 86400) / 3600);
    const m = Math.floor((sec % 3600) / 60);
    return `${d ? `${d}d ` : ''}${pad(h)}:${pad(m)}:${pad(sec % 60)}`;
  }

  const api = { bytes, rate, pct, uptime };
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  else root.fmt = api;
})(this);
```

`src/winCounters.js`:

```js
const LUID_RE = /luid_0x[0-9a-f]{8}_0x[0-9a-f]{8}/;
const EMPTY = { gpu: null, disk: { active: null }, net: { rxSec: null, txSec: null } };

const clamp = (v) => Math.min(100, Math.max(0, v));

function toSnapshot(samples, adapters) {
  const known = new Map(adapters.map((a) => [a.luid.toLowerCase(), a]));
  const gpus = new Map();
  const entry = (luid) => {
    if (!gpus.has(luid)) gpus.set(luid, { util: 0, dedicated: null, shared: null });
    return gpus.get(luid);
  };
  for (const a of adapters) if (!a.software) entry(a.luid.toLowerCase());

  let idle = null;
  let rx = null;
  let tx = null;
  for (const { p, v } of samples) {
    const key = String(p).toLowerCase();
    const luid = (key.match(LUID_RE) || [])[0];
    if (luid && known.get(luid)?.software) continue;
    if (luid && key.includes('\\gpu engine(')) entry(luid).util += v;
    else if (luid && key.endsWith('\\dedicated usage')) entry(luid).dedicated = v;
    else if (luid && key.endsWith('\\shared usage')) entry(luid).shared = v;
    else if (key.endsWith('\\% idle time')) idle = v;
    else if (key.endsWith('\\bytes received/sec')) rx = (rx ?? 0) + v;
    else if (key.endsWith('\\bytes sent/sec')) tx = (tx ?? 0) + v;
  }

  const gpu = [...gpus].map(([luid, g], i) => {
    const a = known.get(luid);
    return {
      name: a ? a.name : `GPU ${i}`,
      util: Math.min(100, g.util),
      dedicated: g.dedicated == null ? null : { used: g.dedicated, total: a ? a.dedicatedTotal : null },
      shared: g.shared == null ? null : { used: g.shared, total: a ? a.sharedTotal : null },
    };
  });

  return {
    gpu,
    disk: { active: idle == null ? null : clamp(100 - idle) },
    net: { rxSec: rx, txSec: tx },
  };
}

function createParser() {
  let adapters = [];
  return (line) => {
    let msg;
    try {
      msg = JSON.parse(line);
    } catch {
      return null;
    }
    if (Array.isArray(msg?.adapters)) {
      adapters = msg.adapters;
      return null;
    }
    if (Array.isArray(msg?.samples)) return toSnapshot(msg.samples, adapters);
    return null;
  };
}

module.exports = { createParser, toSnapshot, EMPTY };
```

```powershell
# Streams JSON lines to stdout for src/winCounters.js:
#   1. once:  {"adapters":[{luid,name,software,dedicatedTotal,sharedTotal}]}
#   2. ~1/s:  {"samples":[{p,v}]}
[Console]::OutputEncoding = [Text.Encoding]::UTF8

function Emit($obj) {
  # Parent gone (broken pipe) -> exit instead of looping forever as an orphan.
  try { [Console]::Out.WriteLine((ConvertTo-Json -Compress -Depth 4 -InputObject $obj)) } catch { exit }
}

$adapters = @()
try {
  Add-Type -TypeDefinition @'
using System;
using System.Collections.Generic;
using System.Runtime.InteropServices;

public static class Dxgi {
  [StructLayout(LayoutKind.Sequential, CharSet = CharSet.Unicode)]
  public struct DESC1 {
    [MarshalAs(UnmanagedType.ByValTStr, SizeConst = 128)] public string Description;
    public uint VendorId, DeviceId, SubSysId, Revision;
    public UIntPtr DedicatedVideoMemory, DedicatedSystemMemory, SharedSystemMemory;
    public uint LuidLow;
    public int LuidHigh;
    public uint Flags;
  }

  [ComImport, Guid("770aae78-f26f-4dba-a829-253c83d1b387"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
  interface IFactory1 {
    // 9 inherited methods (IDXGIObject x4, IDXGIFactory x5) precede EnumAdapters1.
    void m0(); void m1(); void m2(); void m3(); void m4(); void m5(); void m6(); void m7(); void m8();
    [PreserveSig] int EnumAdapters1(uint index, out IAdapter1 adapter);
  }

  [ComImport, Guid("29038f61-3839-4626-91fd-086879011a05"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
  interface IAdapter1 {
    // 7 inherited methods (IDXGIObject x4, IDXGIAdapter x3) precede GetDesc1.
    void m0(); void m1(); void m2(); void m3(); void m4(); void m5(); void m6();
    void GetDesc1(out DESC1 desc);
  }

  [DllImport("dxgi.dll")]
  static extern int CreateDXGIFactory1(ref Guid riid, out IFactory1 factory);

  public static DESC1[] List() {
    var iid = new Guid("770aae78-f26f-4dba-a829-253c83d1b387");
    IFactory1 factory;
    if (CreateDXGIFactory1(ref iid, out factory) != 0) return new DESC1[0];
    var result = new List<DESC1>();
    IAdapter1 adapter;
    for (uint i = 0; factory.EnumAdapters1(i, out adapter) == 0; i++) {
      DESC1 d;
      adapter.GetDesc1(out d);
      result.Add(d);
    }
    return result.ToArray();
  }
}
'@
  $adapters = @([Dxgi]::List() | ForEach-Object {
    @{
      luid           = 'luid_0x{0:x8}_0x{1:x8}' -f $_.LuidHigh, $_.LuidLow
      name           = $_.Description
      software       = [bool]($_.Flags -band 2)
      dedicatedTotal = [double]$_.DedicatedVideoMemory.ToUInt64()
      sharedTotal    = [double]$_.SharedSystemMemory.ToUInt64()
    }
  })
} catch {}
Emit @{ adapters = $adapters }

$paths = @(
  '\GPU Engine(*engtype_3D)\Utilization Percentage',
  '\GPU Adapter Memory(*)\Dedicated Usage',
  '\GPU Adapter Memory(*)\Shared Usage',
  '\PhysicalDisk(_Total)\% Idle Time',
  '\Network Interface(*)\Bytes Received/sec',
  '\Network Interface(*)\Bytes Sent/sec'
)

while ($true) {
  $samples = @()
  try {
    # Get-Counter waits one 1s sample interval, which paces the loop.
    $samples = @((Get-Counter -Counter $paths -ErrorAction SilentlyContinue).CounterSamples |
      ForEach-Object { @{ p = $_.Path; v = $_.CookedValue } })
  } catch {}
  Emit @{ samples = $samples }
  if ($samples.Count -eq 0) { Start-Sleep -Seconds 1 }
}
```

```js
function startCounters({ onSnapshot, spawn = childProcess.spawn, restartMs = 5000, maxFailures = 3 }) {
  const parse = createParser();
  let child = null;
  let timer = null;
  let failures = 0;
  let stopped = false;

  function launch() {
    const proc = spawn(
      'powershell.exe',
      ['-NoProfile', '-NonInteractive', '-ExecutionPolicy', 'Bypass', '-File', SCRIPT_PATH],
      { windowsHide: true },
    );
    child = proc;
    let buf = '';
    proc.stdout.setEncoding('utf8');
    proc.stdout.on('data', (chunk) => {
      const lines = (buf + chunk).split(/\r?\n/);
      buf = lines.pop();
      for (const line of lines) {
        const snap = parse(line);
        if (snap && !stopped) {
          failures = 0;
          onSnapshot(snap);
        }
      }
    });
    // 'exit' may or may not follow 'error'; count the failure once.
    let ended = false;
    const onEnd = () => {
      if (ended || stopped) return;
      ended = true;
      child = null;
      failures += 1;
      if (failures >= maxFailures) onSnapshot(EMPTY);
      else timer = setTimeout(launch, restartMs);
    };
    proc.on('error', onEnd);
    proc.on('exit', onEnd);
  }

  launch();
  return {
    stop() {
      stopped = true;
      clearTimeout(timer);
      if (child) child.kill();
      child = null;
    },
  };
}
```

`src/metrics.js`:

```js
const os = require('os');
const { startCounters } = require('./winCounters');

const FAST_MS = 1000;
const SLOW_MS = 10000;
const MB = 1024 * 1024;
const IS_WINDOWS = process.platform === 'win32';

async function safe(fn) {
  try {
    return await fn();
  } catch {
    return null;
  }
}

function sum(values) {
  const nums = values.filter((v) => typeof v === 'number');
  return nums.length ? nums.reduce((a, b) => a + b, 0) : null;
}

async function fast(si, { windows = IS_WINDOWS } = {}) {
  const [load, nets] = await Promise.all([
    safe(() => si.currentLoad()),
    windows ? null : safe(() => si.networkStats('*')),
  ]);
  const total = os.totalmem();
  const snap = {
    cpu: load
      ? { total: load.currentLoad, cores: (load.cpus || []).map((c) => c.load) }
      : { total: null, cores: [] },
    mem: { used: total - os.freemem(), total },
  };
  if (!windows) {
    snap.net = {
      rxSec: sum((nets || []).map((n) => n.rx_sec)),
      txSec: sum((nets || []).map((n) => n.tx_sec)),
    };
    // ponytail: systeminformation has no disk busy % on macOS/Linux; read /proc/diskstats io_ticks if it is wanted
    snap.disk = { active: null };
  }
  return snap;
}

function toGpu(g) {
  const total = g.memoryTotal ?? g.vram;
  return {
    name: g.model,
    util: g.utilizationGpu ?? null,
    dedicated: g.memoryUsed == null ? null : { used: g.memoryUsed * MB, total: total == null ? null : total * MB },
    shared: null,
  };
}

async function slow(si, { windows = IS_WINDOWS } = {}) {
  const [fs, bat, temp, gfx] = await Promise.all([
    safe(() => si.fsSize()),
    safe(() => si.battery()),
    windows ? null : safe(() => si.cpuTemperature()),
    windows ? null : safe(() => si.graphics()),
  ]);
  const snap = {
    disks: (fs || []).filter((d) => d.size > 0).map((d) => ({ mount: d.mount, used: d.used, size: d.size })),
    battery: bat && bat.hasBattery ? { percent: bat.percent, charging: bat.isCharging } : null,
    tempC: temp && temp.main > 0 ? temp.main : null,
    uptimeSec: os.uptime(),
  };
  if (!windows) snap.gpu = gfx ? gfx.controllers.map(toGpu) : null;
  return snap;
}

function poll(fn, ms, post) {
  let busy = false;
  let stopped = false;
  const tick = async () => {
    if (busy) return;
    busy = true;
    try {
      const value = await fn();
      if (!stopped) post(value);
    } finally {
      busy = false;
    }
  };
  tick();
  const id = setInterval(tick, ms);
  return () => {
    stopped = true;
    clearInterval(id);
  };
}

function startSources({ si, post, windows = IS_WINDOWS, counters = startCounters }) {
  const stops = [
    poll(() => fast(si, { windows }), FAST_MS, post),
    poll(() => slow(si, { windows }), SLOW_MS, post),
  ];
  if (windows) {
    const c = counters({ onSnapshot: post });
    stops.push(() => c.stop());
  }
  return () => stops.forEach((stop) => stop());
}

module.exports = { fast, slow, poll, startSources };
```

```js
class MetricsViewProvider {
  constructor(extensionUri) {
    this.extensionUri = extensionUri;
    this.stopSources = null;
  }

  resolveWebviewView(view) {
    const { webview } = view;
    webview.options = {
      enableScripts: true,
      localResourceRoots: [vscode.Uri.joinPath(this.extensionUri, 'media')],
    };
    webview.html = this.html(webview);
    // The webview is usually recreated when the view is shown again and its script posts 'ready' on
    // every load; visibility also restarts sources in case the content survived. start() is idempotent.
    webview.onDidReceiveMessage((msg) => {
      if (msg === 'ready') this.start(webview);
    });
    view.onDidChangeVisibility(() => {
      if (view.visible) this.start(webview);
      else this.stop();
    });
    view.onDidDispose(() => this.stop());
  }

  start(webview) {
    if (this.stopSources) return;
    this.stopSources = startSources({ si, post: (snap) => webview.postMessage(snap) });
  }

  stop() {
    if (this.stopSources) this.stopSources();
    this.stopSources = null;
  }

  html(webview) {
    const nonce = crypto.randomBytes(16).toString('base64');
    const uri = (file) => webview.asWebviewUri(vscode.Uri.joinPath(this.extensionUri, 'media', file));
    return `<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta http-equiv="Content-Security-Policy" content="default-src 'none'; style-src ${webview.cspSource}; script-src 'nonce-${nonce}';">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<link rel="stylesheet" href="${uri('view.css')}">
</head>
<body>
<div id="root"></div>
<script nonce="${nonce}" src="${uri('format.js')}"></script>
<script nonce="${nonce}" src="${uri('view.js')}"></script>
</body>
</html>`;
  }
}

let provider;

function activate(context) {
  provider = new MetricsViewProvider(context.extensionUri);
  context.subscriptions.push(vscode.window.registerWebviewViewProvider('systemMetrics.view', provider));
}

function deactivate() {
  if (provider) provider.stop();
}

module.exports = { activate, deactivate };
```

```js
  function el(tag, cls, parent, text) {
    const e = document.createElement(tag);
    if (cls) e.className = cls;
    if (text != null) e.textContent = text;
    if (parent) parent.appendChild(e);
    return e;
  }

  function section(title, parent = root) {
    const box = el('section', 'section', parent);
    const head = el('div', 'head', box);
    el('span', 'title', head, title);
    const value = el('span', 'value', head, 'N/A');
    return { box, value };
  }

  function cssVar([name, fallback]) {
    return getComputedStyle(document.documentElement).getPropertyValue(name).trim() || fallback;
  }

  // max: fixed scale (percent graphs) or null to auto-scale (network, floor 1 KB/s).
  function graph(parent, colors, max) {
    const g = { canvas: el('canvas', 'graph', parent), colors, max, hist: colors.map(() => []) };
    graphs.push(g);
    return g;
  }

  function push(g, values) {
    values.forEach((v, i) => {
      const h = g.hist[i];
      h.push(v == null ? 0 : v);
      if (h.length > POINTS) h.shift();
    });
    draw(g);
  }

  function draw(g) {
    const { canvas } = g;
    const dpr = window.devicePixelRatio || 1;
    const w = canvas.clientWidth;
    const h = canvas.clientHeight;
    if (!w || !h) return;
    if (canvas.width !== Math.round(w * dpr) || canvas.height !== Math.round(h * dpr)) {
      canvas.width = Math.round(w * dpr);
      canvas.height = Math.round(h * dpr);
    }
    const ctx = canvas.getContext('2d');
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, w, h);

    ctx.strokeStyle = cssVar(['--vscode-panel-border', 'rgba(128,128,128,0.35)']);
    ctx.lineWidth = 1;
    ctx.beginPath();
    for (let i = 1; i < 4; i++) {
      const y = Math.round((h * i) / 4) + 0.5;
      ctx.moveTo(0, y);
      ctx.lineTo(w, y);
    }
    for (let i = 1; i < 6; i++) {
      const x = Math.round((w * i) / 6) + 0.5;
      ctx.moveTo(x, 0);
      ctx.lineTo(x, h);
    }
    ctx.stroke();

    const max = g.max ?? Math.max(1024, ...g.hist.flat());
    const step = w / (POINTS - 1);
    g.hist.forEach((hist, k) => {
      if (!hist.length) return;
      const color = cssVar(g.colors[k]);
      const x0 = w - (hist.length - 1) * step;
      const y = (v) => h - 1 - (Math.min(v, max) / max) * (h - 2);
      ctx.beginPath();
      hist.forEach((v, i) => (i ? ctx.lineTo(x0 + i * step, y(v)) : ctx.moveTo(x0, y(v))));
      ctx.strokeStyle = color;
      ctx.lineWidth = 1.5;
      ctx.stroke();
      ctx.lineTo(w, h);
      ctx.lineTo(x0, h);
      ctx.closePath();
      ctx.globalAlpha = 0.2;
      ctx.fillStyle = color;
      ctx.fill();
      ctx.globalAlpha = 1;
    });

  }

  const temp = info('Temperature');
  const uptime = info('Uptime');

    cpu.value.textContent = fmt.pct(total);
    push(cpuGraph, [total]);
    if (coreBars.length !== cores.length) {

    push(diskGraph, [active]);
  }

  function renderDrives(list) {
    const key = list.map((d) => d.mount).join('|');
    if (key !== drivesKey) {
      drivesKey = key;
      drivesBox.replaceChildren();
      driveBars = list.map((d) => bar(drivesBox, d.mount));
    }
    list.forEach((d, i) => setBar(driveBars[i], d.used, d.size, `${fmt.bytes(d.used)} / ${fmt.bytes(d.size)}`));
  }

  function renderNet({ rxSec, txSec }) {
    net.value.textContent = `\u2193 ${fmt.rate(rxSec)}  \u2191 ${fmt.rate(txSec)}`;
    push(netGraph, [rxSec, txSec]);
  }

  window.addEventListener('message', ({ data: m }) => {
    if (m.cpu) renderCpu(m.cpu);
    if (m.mem) renderMem(m.mem);
    if ('gpu' in m) renderGpu(m.gpu);
    if (m.disk) renderDisk(m.disk);
    if (m.disks) renderDrives(m.disks);
    if (m.net) renderNet(m.net);
    if ('battery' in m) {
      battery.textContent = m.battery ? `${m.battery.percent}%${m.battery.charging ? ' (charging)' : ''}` : 'N/A';
    }
    if ('tempC' in m) temp.textContent = m.tempC == null ? 'N/A' : `${Math.round(m.tempC)} \u00b0C`;
    if ('uptimeSec' in m) uptime.textContent = fmt.uptime(m.uptimeSec);
  });

  new ResizeObserver(() => graphs.forEach(draw)).observe(root);
  vscode.postMessage('ready');
})();
```
